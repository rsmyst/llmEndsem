\begin{frame}{System Architecture: Three Cohesive Contributions}
  \begin{block}{The MATP Pipeline}
    A modular, end-to-end context management and tool policy pipeline for long-horizon agents:
  \end{block}
  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.32\textwidth}
      \begin{block}{\small 1. Background Abstraction}
        {\footnotesize \textbf{Shishir Hegde (Contrib 2)}}
        \begin{itemize}
          \scriptsize
          \item Intercepts raw tool outputs $O_{\mathrm{raw}}$.
          \item Distills into compact tuples $G$ and $(S,R,O)$ triples.
          \item Zero-dependency subword vector store and episodic graph.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{\small 2. Versioning Filter}
        {\footnotesize \textbf{Rahul Saravanan (Contrib 1)}}
        \begin{itemize}
          \scriptsize
          \item Logical clock $t$ tracking discrete causal mutations.
          \item Piecewise $R=0$ superseding of stale lineage.
          \item Exponential recency decay $e^{-\lambda \Delta t}$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{\small 3. Predictive Tool Router}
        {\footnotesize \textbf{Ashab Muhammed (Contrib 3)}}
        \begin{itemize}
          \scriptsize
          \item Bayesian success ranking $p(\mathrm{tool}|\mathrm{goal})$.
          \item Parameter auto-correction hints.
          \item Pre-fetches high-confidence tools before deliberation.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The long horizon problem}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{block}{What breaks today}
        \begin{itemize}
          \item Agent keeps 10 to 30 tool outputs in context. FIFO eviction drops early, critical state and causes catastrophic forgetting.
          \item Naive RAG keeps everything but has no notion of mutation. A query still matches schema v1 on keyword overlap and causes stale state hallucination.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{What we need}
        \begin{itemize}
          \item Keep all history, but retrieve only the current version of each resource.
          \item No token bloat, no external service, no network.
        \end{itemize}
      \end{block}
      \vspace{0.6em}
      {\small \textbf{Question:} how to retain history without returning superseded state?}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Objectives}
  \begin{block}{Goal}
    Replace rigid context pruning with a persistent, queryable semantic memory buffer exposed as an MCP server.
  \end{block}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Dual memory}
        \begin{itemize}
          \item Working memory: small active context for reasoning.
          \item Long term buffer: keeps every version, tagged by resource id and logical clock.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Operations}
        \begin{itemize}
          \item Ingest: tag, embed, version, mark prior version superseded.
          \item Retrieve: score with $R(q,v_i)$, filter stale, decay by age.
          \item Audit: full history per resource.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Scoring: $R(q,v_i)$}
  \begin{block}{State aware retrieval}
    Let $q$ be the query embedding, $v_i$ a stored state, $t_{\mathrm{curr}}$ the logical clock, $t_i$ the birth step of $v_i$:
    \[
      R(q,v_i)=
      \begin{cases}
        0 & \text{if }\exists j: \text{id}_j=\text{id}_i \land t_j>t_i\\[2pt]
        \cos(q,v_i)\,\cdot e^{-\lambda\,(t_{\mathrm{curr}}-t_i)} & \text{otherwise}
      \end{cases}
    \]
  \end{block}
  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.33\textwidth}
      \textbf{1. Logical clock $t$}\\[1pt]
      {\footnotesize Discrete causal steps, not wall time. Each ingest increments the clock.}
    \end{column}
    \begin{column}{0.33\textwidth}
      \textbf{2. Superseding $R=0$}\\[1pt]
      {\footnotesize If a newer version of the same resource exists, the older is zeroed when filtering is enabled.}
    \end{column}
    \begin{column}{0.33\textwidth}
      \textbf{3. Decay $e^{-\lambda\Delta t}$}\\[1pt]
      {\footnotesize Among active states, recent steps outrank old ones. Default $\lambda=0.05$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Benchmark: schema refactor ($t=1$ to $16$)}
  {\footnotesize Setup: agent ingests db/schema.sql v1, configs, src/models/user.py v1 which depends on schema v1, then schema v2 and auth v2. Clock advanced to $t_{\mathrm{curr}}=16$. Query mentions both password hash from v1 and oauth from v2 so lexical overlap favours stale text.}

  \vspace{0.4em}
  \begin{table}
    \centering
    \footnotesize
    \begin{tabular}{lccc}
      \toprule
      Method & Lineage stale in top $k$ & Effective precision & Retention\\
      \midrule
      FIFO (drop old turns) & & 0\% (forgotten) & 0\%\\
      Naive RAG (cosine only, $k=5$) & 1/5 = 20\% & 3/5 = 60\% & 100\%\\
      \textbf{MATP (this work, $k=4$, $\lambda=0.05$)} & \textbf{0/4 = 0\%} & \textbf{3/4 = 75\%} & \textbf{100\%}\\
      \bottomrule
    \end{tabular}
  \end{table}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\footnotesize \textbf{What the numbers mean}}
      \begin{itemize}
        \footnotesize
        \item MATP eliminates direct lineage stale: schema v1 is $R=0$ and never returned.
        \item Dependent leak remains: src/models/user.py v1 still scores $R\approx0.38$ because its resource id differs. So 1/4 is effectively stale.
        \item Naive also surfaces that ORM file, so 2/5 are effectively stale if counting semantic staleness.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\footnotesize \textbf{Decay sweep} (query: configuration parameters)}
      \begin{itemize}
        \footnotesize
        \item $\lambda=0.00$: $R=\cos$ only, older configs tie.
        \item $\lambda=0.05$: recent auth v2 outranks older by factor 0.61.
        \item $\lambda=0.20$: strong recency, top score factor 0.14 for tight horizons.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contribution 2: Background Abstraction Engine (Shishir Hegde)}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{The Bloat and Noise Bottleneck}
        \begin{itemize}
          \footnotesize
          \item Raw tool outputs $O_{\mathrm{raw}}$ (SQL JSON dumps, 100-line stack traces, API payloads, CLI logs) have \textbf{low Signal-to-Noise Ratio (SNR)}.
          \item Storing raw text causes rapid context overflow, token bloat, and noise dilution during semantic retrieval.
        \end{itemize}
      \end{block}
      \begin{block}{Mathematical Formulation}
        \footnotesize
        Extract core state facts into structured tuple:
        \[
          G = (g,\, t,\, p,\, s,\, \sigma,\, \text{parent})
        \]
        \[
          \mathcal{T} = \{(S_i, R_i, O_i)\} \quad \Longrightarrow \quad \mathrm{Size}(G) \ll \mathrm{Size}(O_{\mathrm{raw}})
        \]
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Architecture and Execution Hook}
        \begin{itemize}
          \footnotesize
          \item \textbf{Non-Blocking Ingest:} Hooks directly into post-tool execution loop via \texttt{matp\_index\_trajectory}.
          \item \textbf{Deterministic Vectorization:} Subword $n$-gram hash embedder with term frequency saturation (zero-infra, CPU-local).
          \item \textbf{Episodic Triples Extraction:} Extracts $(S, R, O)$ facts and entity mentions at ingestion.
          \item \textbf{SQLite Durability:} WAL-backed store replaces per-index JSON rewrites; legacy JSON imported once.
          \item \textbf{Causal DAG + Macros:} Soft validity labels and reusable 10-step composite skills.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

          \item Filters 87.4\% -- 94.1\% low-SNR boilerplate (headers, stack frames, OCR noise).
          \item Strict P@3 rises from 60\% to 80\%; policy weights suppress stale rows.
        \end{itemize}
      \end{block}

        \begin{itemize}

        \begin{itemize}
          \footnotesize

\begin{frame}{Takeaway}
  \begin{itemize}
    \item \textbf{Retain without bloat (Contrib 2):} Background abstraction achieves 92.6\% token compression (13.5$\times$), preserving full 30-turn history in 4,524 tokens with 88\% equal-budget fact retention.
    \item \textbf{Retrieve without stale (Contrib 2):} Causal DAG labels 24/24 replay nodes correctly; stale and superseded rows are down-weighted or excluded.
    \item \textbf{Reuse without rediscovery (Contrib 2):} Held-out macros replay with 100\% sequence and slot-binding accuracy.
    \item \textbf{Route with confidence (Contrib 3):} Laplace-smoothed predictive pre-fetching recommends optimal tools and auto-corrects parameters.
    \item \textbf{Zero external infra:} Local subword vectorizer, SQLite persistence, optional local Ollama worker, and stdio MCP.
  \end{itemize}
